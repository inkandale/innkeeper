# Web UI Settings Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add the Plugins, Access, and Appearance tabs to Settings, with plugin instance config forms rendered from each manifest's JSON Schema.

**Architecture:** Frontend only. The API already has every endpoint: `/plugins`, `/instances`, `/tokens`, `/grants`, `/settings`, and `/themes/{name}.css`. Each tab is one component under `web/src/settings/`. One `SchemaForm` component renders a config schema recursively. A small `theme/link.ts` owns the theme `<link>`, so Appearance can preview a theme by repointing it.

**Tech Stack:** React, React Router, TanStack Query, `openapi-fetch`, Vitest, Testing Library.

**Spec:** `docs/superpowers/specs/2026-10-05-web-ui-design.md`, "Settings" and "Themes". Build order phase 6. The Jobs tab shipped in phase 3.

---

## Design notes

- **Plugin status is instance status.** Plugins have no state of their own; an instance is OK or broken. Each instance row shows its `broken_error`. Saving an instance clears its broken state, so the form says so.
- **The schema form covers what plugins ship.** `string`, `number`, `integer`, `boolean`, string `enum`, and `object` with `properties`. Anything else, and any value whose type doesn't match its schema, is a raw JSON textarea. Keywords like `if`/`then` are ignored; the plugin reports bad config, and the instance shows as broken.
- **Empty means absent.** Clearing a text or number input removes its key, so `required` keeps its meaning. Required fields use the native `required` attribute, and the browser blocks the submit.
- **Config is sent only when it changed.** Any `config`, `secrets`, or `owner_subject` in a PATCH ends the instance's login sessions. Renaming a login provider shouldn't log everyone out.
- **Secrets are write-only.** Each shows "Set" or "Not set", a password input to replace it, and Remove when set. Only typed or removed secrets are sent.
- **Owner subject** shows for `auth` plugins, and **Bind exact matches** for `pull` plugins on edit (the create body has no such field).
- **Theme preview** repoints the theme `<link>` at `/api/themes/{name}.css`, and disables it for the default theme. Leaving the tab puts the saved theme back.
- **Out of this plan:** server-side config validation, an unsaved-edits prompt on the instance form, and plugin install or removal (plugins are folders on disk).

```ts
/** themeLink returns the <link> that loads the selected theme, adding it on first use. */
export function themeLink(): HTMLLinkElement {
  let link = document.getElementById("theme") as HTMLLinkElement | null;
  if (!link) {
    link = document.createElement("link");
    link.id = "theme";
    link.rel = "stylesheet";
    link.href = "/api/theme.css";
    // Appended after the app's CSS, so the theme's rules win.
    document.head.append(link);
  }
  return link;
}

/** showTheme loads name's stylesheet, or turns the theme off for the default (""). */
export function showTheme(name: string) {
  const link = themeLink();
  link.disabled = name === "";
  if (name) link.href = `/api/themes/${encodeURIComponent(name)}.css`;
}
```

```tsx
import { useState, type ReactNode } from "react";

/** Schema is the part of JSON Schema the form renders. */
export type Schema = {
  type?: string;
  title?: string;
  description?: string;
  enum?: unknown[];
  properties?: Record<string, Schema>;
  required?: string[];
};

type Props = {
  schema: Schema;
  /** label is the property name, used when the schema has no title. */
  label: string;
  value: unknown;
  onChange: (v: unknown) => void;
  required?: boolean;
};

const fits: Record<string, (v: unknown) => boolean> = {
  string: (v) => typeof v === "string",
  number: (v) => typeof v === "number",
  integer: (v) => Number.isInteger(v),
  boolean: (v) => typeof v === "boolean",
  object: (v) => typeof v === "object" && v !== null && !Array.isArray(v),
};

/**
 * SchemaForm renders inputs for a JSON Schema: string, number, integer,
 * boolean, string enum, and objects with properties. Anything else is raw JSON.
 */
export function SchemaForm({ schema, label, value, onChange, required }: Props) {
  const title = schema.title ?? label;
  const t = schema.type ?? "";
  // A stored value of another type is shown as JSON rather than lost.
  const known = t in fits && (value === undefined || fits[t](value));

  if (known && t === "object" && schema.properties) {
    const obj = (value ?? {}) as Record<string, unknown>;
    return (
      <fieldset className="schema-object">
        {title && <legend>{title}</legend>}
        {schema.description && <small className="muted">{schema.description}</small>}
        {Object.entries(schema.properties).map(([k, s]) => (
          <SchemaForm
            key={k}
            schema={s}
            label={k}
            value={obj[k]}
            required={schema.required?.includes(k)}
            onChange={(v) => {
              const { [k]: _, ...rest } = obj;
              onChange(v === undefined ? rest : { ...rest, [k]: v });
            }}
          />
        ))}
      </fieldset>
    );
  }

  let control: ReactNode;
  if (known && t === "string" && schema.enum?.every((e) => typeof e === "string")) {
    control = (
      <select className="input" required={required} value={(value as string) ?? ""} onChange={(e) => onChange(e.target.value || undefined)}>
        <option value="">—</option>
        {(schema.enum as string[]).map((e) => (
          <option key={e}>{e}</option>
        ))}
      </select>
    );
  } else if (known && t === "string") {
    control = (
      <input className="input" required={required} value={(value as string) ?? ""} onChange={(e) => onChange(e.target.value || undefined)} />
    );
  } else if (known && (t === "number" || t === "integer")) {
    control = (
      <input
        className="input"
        type="number"
        step={t === "integer" ? 1 : "any"}
        required={required}
        value={(value as number) ?? ""}
        onChange={(e) => onChange(e.target.value === "" ? undefined : Number(e.target.value))}
      />
    );
  } else if (known && t === "boolean") {
    control = <input type="checkbox" checked={value === true} onChange={(e) => onChange(e.target.checked)} />;
  } else {
    control = <RawJson value={value} onChange={onChange} required={required} />;
  }
  return (
    <div className="form-field">
      <label className={t === "boolean" && known ? "check" : undefined}>
        <span>{title}</span>
        {control}
      </label>
      {schema.description && <small className="muted">{schema.description}</small>}
    </div>
  );
}

/** RawJson edits any value as JSON. Text that doesn't parse marks the field invalid and isn't passed on. */
function RawJson({ value, onChange, required }: { value: unknown; onChange: (v: unknown) => void; required?: boolean }) {
  const [text, setText] = useState(value === undefined ? "" : JSON.stringify(value, null, 2));
  return (
    <textarea
      className="input"
      rows={3}
      required={required}
      value={text}
      onChange={(e) => {
        const box = e.target;
        setText(box.value);
        box.setCustomValidity("");
        if (box.value.trim() === "") return onChange(undefined);
        try {
          onChange(JSON.parse(box.value));
        } catch {
          box.setCustomValidity("Not valid JSON");
        }
      }}
    />
  );
}
```

```tsx
import { useMutation, useQuery, useQueryClient } from "@tanstack/react-query";
import { Link } from "react-router";
import { api, must, mustOk } from "../api/client";
import type { components } from "../api/schema";

export type Plugin = components["schemas"]["PluginOut"];
export type InstanceInfo = components["schemas"]["InstanceInfo"];

export const pluginsQuery = { queryKey: ["plugins"], queryFn: async () => must(await api.GET("/plugins")) };

export function Plugins() {
  const qc = useQueryClient();
  const plugins = useQuery(pluginsQuery);
  const instances = useQuery({ queryKey: ["instances"], queryFn: async () => must(await api.GET("/instances")) });
  const remove = useMutation({
    mutationFn: async (id: number) => mustOk(await api.DELETE("/instances/{id}", { params: { path: { id } } })),
    onSuccess: () => qc.invalidateQueries({ queryKey: ["instances"] }),
  });
  const failed = remove.error ?? plugins.error ?? instances.error;
  const all = instances.data ?? [];
  const names = new Set((plugins.data ?? []).map((p) => p.name));
  const orphans = all.filter((i) => !names.has(i.plugin));

  const table = (list: InstanceInfo[]) =>
    list.length > 0 && (
      <table>
        <tbody>
          {list.map((i) => (
            <tr key={i.id}>
              <td>
                <Link to={`/settings/instances/${i.id}`}>{i.name}</Link>
              </td>
              <td>
                {i.broken_error ? (
                  <span className="state failed">
                    Broken: <span>{i.broken_error}</span>
                  </span>
                ) : (
                  <span className="state succeeded">OK</span>
                )}
              </td>
              <td>
                <button
                  className="button"
                  aria-label={`Delete ${i.name}`}
                  onClick={() => confirm(`Delete ${i.name}? Its jobs stay.`) && remove.mutate(i.id)}
                >
                  Delete
                </button>
              </td>
            </tr>
          ))}
        </tbody>
      </table>
    );

  return (
    <section className="plugins">
      {failed && <p className="error">{failed.message}</p>}
      {(plugins.data ?? []).map((p) => (
        <article key={p.name} className="plugin">
          <header>
            <h2>{p.name}</h2>
            <span className="muted">{p.version}</span>
            <ul className="caps">
              {(p.capabilities ?? []).map((c) => (
                <li key={c}>{c}</li>
              ))}
            </ul>
            {p.capabilities?.includes("theme") ? (
              <Link to="/settings/appearance">Select in Appearance</Link>
            ) : (
              <Link className="button" to={`/settings/plugins/${encodeURIComponent(p.name)}/new`}>
                Add instance
              </Link>
            )}
          </header>
          {table(all.filter((i) => i.plugin === p.name))}
        </article>
      ))}
      {orphans.length > 0 && (
        <article className="plugin">
          <header>
            <h2>Not installed</h2>
            <span className="muted">These instances' plugins are gone.</span>
          </header>
          {table(orphans)}
        </article>
      )}
    </section>
  );
}
```

```tsx
/** Instance adds an instance at /settings/plugins/:plugin/new and edits one at /settings/instances/:id. */
export function Instance() {
  const { plugin: pluginName, id } = useParams();
  const plugins = useQuery(pluginsQuery);
  const inst = useQuery({
    queryKey: ["instance", Number(id)],
    enabled: id !== undefined,
    queryFn: async () => must(await api.GET("/instances/{id}", { params: { path: { id: Number(id) } } })),
  });
  const failed = plugins.error ?? inst.error;
  if (failed) return <p className="error">{failed.message}</p>;
  if (plugins.isPending || (id !== undefined && inst.isPending)) return null;
  const name = inst.data?.plugin ?? pluginName;
  const plugin = plugins.data.find((p) => p.name === name);
  if (!plugin) return <p className="error">No plugin named {name} is installed.</p>;
  return <InstanceForm key={inst.data?.id ?? name} plugin={plugin} instance={inst.data} />;
}

function InstanceForm({ plugin, instance }: { plugin: Plugin; instance?: InstanceInfo }) {
  const qc = useQueryClient();
  const navigate = useNavigate();
  const toast = useToast();
  const caps = plugin.capabilities ?? [];
  const [config, setConfig] = useState<unknown>(instance?.config ?? {});
  // Only typed secrets (a string) and removed ones (null) are sent.
  const [secrets, setSecrets] = useState<Record<string, string | null>>({});

  const save = useMutation({
    mutationFn: async (form: FormData) => {
      const name = String(form.get("name")).trim();
      const cfg = (config ?? {}) as Record<string, unknown>;
      const owner = caps.includes("auth") ? String(form.get("owner_subject") ?? "").trim() : undefined;
      const hasSecrets = Object.keys(secrets).length > 0;
      if (!instance) {
        return must(
          await api.POST("/instances", {
            body: {
              plugin: plugin.name,
              name,
              config: cfg,
              secrets: hasSecrets ? (secrets as Record<string, string>) : undefined,
              owner_subject: owner || undefined,
            },
          }),
        );
      }
      // Changing config, secrets, or owner ends the instance's logins, so send them only when they changed.
      return must(
        await api.PATCH("/instances/{id}", {
          params: { path: { id: instance.id } },
          body: {
            name,
            config: JSON.stringify(cfg) !== JSON.stringify(instance.config) ? cfg : undefined,
            secrets: hasSecrets ? secrets : undefined,
            owner_subject: owner !== undefined && owner !== (instance.owner_subject ?? "") ? owner : undefined,
            auto_bind_exact: caps.includes("pull") ? form.get("auto_bind_exact") === "on" : undefined,
          },
        }),
      );
    },
    onSuccess: (saved) => {
      qc.setQueryData(["instance", saved.id], saved);
      qc.invalidateQueries({ queryKey: ["instances"] });
      toast({ text: `Saved ${saved.name}.` });
      navigate("/settings/plugins");
    },
  });

  return (
    <form
      className="instance-form"
      onSubmit={(e) => {
        e.preventDefault();
        save.mutate(new FormData(e.currentTarget));
      }}
    >
      <h2>{instance ? instance.name : `New ${plugin.name} instance`}</h2>
      {instance?.broken_error && (
        <p className="error">
          Broken since {new Date(instance.broken_at ?? instance.updated_at).toLocaleString()}: {instance.broken_error}.
          Saving clears this and restarts the plugin.
        </p>
      )}
      {instance && caps.includes("auth") && (
        <p className="muted">
          Changing the owner, config, or secrets ends every login through this provider, including yours if you used it.
        </p>
      )}
      <div className="form-field">
        <label>
          <span>Name</span>
          <input className="input" name="name" required defaultValue={instance?.name} />
        </label>
      </div>
      {plugin.config && <SchemaForm schema={plugin.config as Schema} label="" value={config} onChange={setConfig} />}
      {(plugin.secrets ?? []).map((s) => {
        const set = instance?.secrets?.includes(s) ?? false;
        const pending = secrets[s];
        return (
          <div className="form-field" key={s}>
            <label>
              <span>{s}</span>
              <input
                className="input"
                type="password"
                autoComplete="new-password"
                value={pending ?? ""}
                onChange={(e) =>
                  setSecrets(({ [s]: _, ...rest }) => (e.target.value ? { ...rest, [s]: e.target.value } : rest))
                }
              />
            </label>
            <small className="muted">{pending === null ? "Removed when you save." : set ? "Set. Type to replace it." : "Not set."}</small>
            {set && pending === undefined && (
              <button type="button" className="button" aria-label={`Remove ${s}`} onClick={() => setSecrets((x) => ({ ...x, [s]: null }))}>
                Remove
              </button>
            )}
          </div>
        );
      })}
      {caps.includes("auth") && (
        <div className="form-field">
          <label>
            <span>Owner subject</span>
            <input className="input" name="owner_subject" defaultValue={instance?.owner_subject} />
          </label>
          <small className="muted">The login that becomes the owner. Leave empty for none.</small>
        </div>
      )}

          <input type="checkbox" name="auto_bind_exact" defaultChecked={instance.auto_bind_exact} /> Bind exact matches when
          reconciling
        </label>
      )}
      {save.error && <p className="error">{save.error.message}</p>}

        </button>

        </Link>
```

```tsx
import { useMutation, useQuery, useQueryClient } from "@tanstack/react-query";
import { useState } from "react";
import { api, must, mustOk } from "../api/client";
import type { components } from "../api/schema";

type Tier = components["schemas"]["Token"]["tier"];
type Created = components["schemas"]["CreatedToken"];
const tiers: Tier[] = ["read", "write", "destructive"];

const when = (t?: string) => (t ? new Date(t).toLocaleString() : "never");

export function Access() {
  const qc = useQueryClient();
  const me = useQuery({ queryKey: ["me"], queryFn: async () => must(await api.GET("/auth/me")) });
  const tokens = useQuery({ queryKey: ["tokens"], queryFn: async () => must(await api.GET("/tokens")) });
  const grants = useQuery({ queryKey: ["grants"], queryFn: async () => must(await api.GET("/grants")) });
  const [created, setCreated] = useState<Created | null>(null);

  const create = useMutation({
    mutationFn: async (body: { name: string; tier: Tier }) => must(await api.POST("/tokens", { body })),
    onSuccess: (t) => {
      setCreated(t);
      return qc.invalidateQueries({ queryKey: ["tokens"] });
    },
  });
  const revokeToken = useMutation({
    mutationFn: async (id: number) => mustOk(await api.DELETE("/tokens/{id}", { params: { path: { id } } })),
    onSuccess: () => qc.invalidateQueries({ queryKey: ["tokens"] }),
  });
  const revokeGrant = useMutation({
    mutationFn: async (id: number) => mustOk(await api.DELETE("/grants/{id}", { params: { path: { id } } })),
    onSuccess: () => qc.invalidateQueries({ queryKey: ["grants"] }),
  });

  // A token can't have more access than the credentials creating it.
  const allowed = tiers.slice(0, tiers.indexOf((me.data?.tier ?? "read") as Tier) + 1);
  const failed = create.error ?? revokeToken.error ?? revokeGrant.error ?? tokens.error ?? grants.error;

  return (

      {failed && <p className="error">{failed.message}</p>}
      <h2>API tokens</h2>

          ))}

          Create
        </button>

        <div className="created" role="status">

          <code>{created.secret}</code>
          <button type="button" className="button" onClick={() => navigator.clipboard?.writeText(created.secret)}>
            Copy
          </button>
          <button type="button" className="button" onClick={() => setCreated(null)}>
            Done
          </button>
        </div>
      )}
      <table>
        <thead>
          <tr>
            <th>Name</th>
            <th>Tier</th>
            <th>Created</th>
            <th>Last used</th>
            <th />
```

```tsx
import { useMutation, useQuery, useQueryClient } from "@tanstack/react-query";
import { useEffect, useState } from "react";
import { api, must } from "../api/client";
import { showTheme } from "../theme/link";
import { pluginsQuery } from "./Plugins";

export function Appearance() {
  const qc = useQueryClient();
  const plugins = useQuery(pluginsQuery);
  const settings = useQuery({ queryKey: ["settings"], queryFn: async () => must(await api.GET("/settings")) });
  const [picked, setPicked] = useState<string | null>(null);
  const saved = settings.data?.theme ?? "";
  const shown = picked ?? saved;

  // Preview the pick. The cleanup puts the saved theme back when the pick changes or the tab closes.
  const ready = settings.data !== undefined;

    if (!ready) return;
    showTheme(shown);
    return () => showTheme(saved);
  }, [ready, shown, saved]);

  const save = useMutation({
    mutationFn: async (theme: string) => must(await api.PATCH("/settings", { body: { theme } })),
    onSuccess: (s) => {
      qc.setQueryData(["settings"], s);
      setPicked(null);
    },
  });

  const themes = (plugins.data ?? []).filter((p) => p.capabilities?.includes("theme")).map((p) => p.name);
  const failed = save.error ?? settings.error ?? plugins.error;
  const changed = picked !== null && picked !== saved;

      <fieldset>
        <legend>Theme</legend>
        {["", ...themes].map((name) => (

          </label>
        ))}

          Save

            Revert
          </button>
```
