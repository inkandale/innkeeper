> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A fresh server can be claimed from the browser with a setup code from the log, or from config with `INNKEEPER_OWNER_PASSWORD_HASH`.

**Architecture:** Two optional protocol methods (`auth.setup`, `auth.setup.complete`) let an auth plugin turn a setup form into instance secrets. The API keeps a setup code in memory while no login provider exists, serves `GET`/`POST /api/setup`, and serializes claims with logins through the existing `loginSlot`. The web UI gains a `/setup` page that the login page redirects to when the server is unclaimed. `password-login` in innkeeper-plugins implements the methods.

**Tech Stack:** Go (Huma, pgx), React + TanStack Query + openapi-fetch, Vitest.

```go
// Optional auth methods that let the owner claim a new server. auth.setup
// returns a form; auth.setup.complete turns its fields into the secrets of
// the instance the core creates. Both run before any instance exists, with
// an empty config and no secrets.
const (
	MethodAuthSetup         = "auth.setup"
	MethodAuthSetupComplete = "auth.setup.complete"
)

// AuthSetupResult is the setup form, a JSON Schema object like
// AuthBeginResult.Form.
type AuthSetupResult struct {
	Form json.RawMessage `json:"form"`
}

// AuthSetupCompleteParams carries the setup form's fields.
type AuthSetupCompleteParams struct {
	Params map[string]string `json:"params"`
}

// AuthSetupCompleteResult is the new instance's secrets and the subject
// auth.complete will report for the owner.
type AuthSetupCompleteResult struct {
	Secrets map[string]string `json:"secrets"`
	Subject string            `json:"subject"`
}
```

```go
package api

import (
	"context"
	"crypto/rand"
	"crypto/subtle"
	"errors"
	"fmt"
	"regexp"
	"strings"
	"unicode"

	"github.com/davidgibbons/innkeeper/internal/plugins"
)

// bcryptHash matches the shape of a bcrypt hash, as password-login makes.
var bcryptHash = regexp.MustCompile(`^\$2[aby]\$(0[4-9]|[12][0-9]|3[01])\$[./A-Za-z0-9]{53}$`)

// claimed reports whether the server has a usable login provider.
func (s *Server) claimed(ctx context.Context) (bool, error) {
	list, err := s.loginProviders(ctx)
	return len(list) > 0, err
}

// Bootstrap runs at startup. On an unclaimed server it creates the owner's
// password login from ownerHash, or without one, logs a setup code.
func (s *Server) Bootstrap(ctx context.Context, ownerHash string) error {
	claimed, err := s.claimed(ctx)
	if err != nil {
		return err
	}
	if claimed {
		if ownerHash != "" {
			s.Log.Warn("INNKEEPER_OWNER_PASSWORD_HASH is ignored: the server already has a login provider")
		}
		return nil
	}
	if ownerHash == "" {
		s.ensureSetupCode()
		return nil
	}
	if !bcryptHash.MatchString(ownerHash) {
		return errors.New("INNKEEPER_OWNER_PASSWORD_HASH is not a bcrypt hash; make one with: password-login hash")
	}
	owner := "owner"
	if _, err := s.Registry.CreateInstance(ctx, plugins.NewInstance{Plugin: "password-login", Name: "Login",
		Secrets: map[string]string{"password_hash": ownerHash}, OwnerSubject: &owner}); err != nil {
		return fmt.Errorf("create the owner's login from INNKEEPER_OWNER_PASSWORD_HASH: %w", err)
	}
	s.Log.Info("created the owner's login from INNKEEPER_OWNER_PASSWORD_HASH")
	return nil
}

// ensureSetupCode returns the setup code, making and logging one if needed.
func (s *Server) ensureSetupCode() string {
	s.setupMu.Lock()
	defer s.setupMu.Unlock()
	if s.setupCode == "" {
		s.setupCode = groupCode(rand.Text())
		s.Log.Warn("server is unclaimed; open /setup and enter the setup code", "code", s.setupCode)
	}
	return s.setupCode
}

func (s *Server) currentSetupCode() string {
	s.setupMu.Lock()
	defer s.setupMu.Unlock()
	return s.setupCode
}

func (s *Server) clearSetupCode() {
	s.setupMu.Lock()
	s.setupCode = ""
	s.setupMu.Unlock()
}

// groupCode splits code into groups of four for typing.
func groupCode(code string) string {
	var groups []string
	for len(code) > 4 {
		groups = append(groups, code[:4])
		code = code[4:]
	}
	return strings.Join(append(groups, code), "-")
}

// normalizeCode drops dashes and spaces and upper-cases the rest, so a
// typed code matches however it was grouped.
func normalizeCode(code string) string {
	return strings.Map(func(r rune) rune {
		if r == '-' || unicode.IsSpace(r) {
			return -1
		}
		return unicode.ToUpper(r)
	}, code)
}

// codeMatches compares a typed code with the setup code in constant time.
func codeMatches(want, got string) bool {
	return want != "" && subtle.ConstantTimeCompare([]byte(normalizeCode(want)), []byte(normalizeCode(got))) == 1
}
```

```go
type setupOut struct{ Body []setupPlugin }

// setupInstance runs an auth plugin before it has an instance.
func setupInstance(m *protocol.Manifest) pluginhost.Instance {
	return pluginhost.Instance{ID: "setup-" + m.Name, Version: "0", Manifest: m,
		Params: protocol.InitializeParams{Config: map[string]any{}, Secrets: map[string]string{}}}
}

func (s *Server) setupRoutes(api huma.API) {
	tags := []string{"Auth"}
	huma.Register(api, huma.Operation{OperationID: "get-setup", Method: http.MethodGet, Path: "/setup",
		Summary: "List the ways to claim this server",
		Description: "Lists the login plugins that can claim a server with no login provider. 404 once the server is claimed.",
		Tags:        tags, Metadata: public, Security: noAuth},
		func(ctx context.Context, _ *struct{}) (*setupOut, error) {
			claimed, err := s.claimed(ctx)
			if err != nil {
				return nil, s.fail(err)
			}
			if claimed {
				return nil, huma.Error404NotFound("the server is already set up")
			}
			s.ensureSetupCode()
			out := &setupOut{Body: []setupPlugin{}}
			manifests := s.Registry.Manifests()
			names := slices.Sorted(maps.Keys(manifests))
			for _, name := range names {
				m := manifests[name]
				if !slices.Contains(m.Capabilities, "auth") {
					continue
				}
				var res protocol.AuthSetupResult
				err := s.Plugins.Call(ctx, setupInstance(m), protocol.MethodAuthSetup, nil, &res)
				var rpcErr *protocol.Error
				if errors.As(err, &rpcErr) && rpcErr.Code == protocol.CodeMethodNotFound {
					continue
				}
				if err != nil {
					s.Log.Warn("auth.setup failed", "plugin", name, "err", err)
					continue
				}
				out.Body = append(out.Body, setupPlugin{Plugin: name, Form: res.Form})
			}
			return out, nil
		})

	huma.Register(api, huma.Operation{OperationID: "claim-server", Method: http.MethodPost, Path: "/setup",
		Summary: "Claim this server",
		Description: "Creates the owner's login provider with the setup code from the server log, and sets the session cookie.",
		Tags:        tags, Metadata: public, Security: noAuth, DefaultStatus: http.StatusNoContent},
		func(ctx context.Context, in *claimIn) (*cookiesOut, error) {
			select {
			case s.loginSlot <- struct{}{}:
				defer func() { <-s.loginSlot }()
			case <-ctx.Done():
				return nil, huma.Error503ServiceUnavailable("login busy, try again")
			}
			// One answer for a wrong code and a claimed server, so the
			// page reveals neither.
			refuse := func() error {
				time.Sleep(failedLoginDelay)
				return huma.Error403Forbidden("setup failed; check the setup code in the server log")
			}
			if !codeMatches(s.currentSetupCode(), in.Body.Code) {
				return nil, refuse()
			}
			if claimed, err := s.claimed(ctx); err != nil {
				return nil, s.fail(err)
			} else if claimed {
				return nil, refuse()
			}
			m := s.Registry.Manifests()[in.Body.Plugin]
			if m == nil || !slices.Contains(m.Capabilities, "auth") {
				return nil, huma.Error400BadRequest("no such login plugin: " + in.Body.Plugin)
			}
			var res protocol.AuthSetupCompleteResult
			err := s.Plugins.Call(ctx, setupInstance(m), protocol.MethodAuthSetupComplete,
				protocol.AuthSetupCompleteParams{Params: in.Body.Params}, &res)
			var rpcErr *protocol.Error
			if errors.As(err, &rpcErr) {
				return nil, huma.Error400BadRequest(rpcErr.Message)
			}
			if err != nil {
				return nil, s.fail(err)
			}
			if res.Subject == "" {
				return nil, huma.Error502BadGateway("plugin: auth.setup.complete returned no subject")
			}
			id, err := s.Registry.CreateInstance(ctx, plugins.NewInstance{Plugin: m.Name, Name: "Login",
				Secrets: res.Secrets, OwnerSubject: &res.Subject})
			if err != nil {
				return nil, s.fail(err)
			}
			s.clearSetupCode()
			s.Log.Info("server claimed", "plugin", m.Name, "instance", id)
			secret, err := s.Auth.CreateSession(ctx, id, res.Subject)
			if err != nil {
				return nil, s.fail(err)
			}
			return &cookiesOut{SetCookie: []*http.Cookie{s.cookie(sessionCookie, secret, auth.SessionTTL)}}, nil
		})
```

```tsx
import { useQuery } from "@tanstack/react-query";
import { useState, type FormEvent } from "react";
import { Link, useNavigate } from "react-router";
import { api, ApiError, must } from "../api/client";
import { formFields } from "./fields";
import "./Login.css";

/** Setup claims a server that has no login provider yet. */
export function Setup() {
  const setup = useQuery({ queryKey: ["setup"], queryFn: async () => must(await api.GET("/setup")) });
  const [chosen, setChosen] = useState<string | null>(null);
  const list = setup.data ?? [];
  const plugin = list.find((p) => p.plugin === chosen) ?? (list.length === 1 ? list[0] : undefined);

  let body;
  if (setup.isPending) body = null;
  else if (setup.error instanceof ApiError && setup.error.status === 404)
    body = (
      <p>
        This server is already set up. <Link to="/login">Log in</Link>
      </p>
    );
  else if (setup.isError) body = <p className="error">{setup.error.message}</p>;
  else if (list.length === 0)
    body = <p>No installed login plugin can claim a server. The README's "First login" section explains how to add one.</p>;
  else if (!plugin)
    body = list.map((p) => (
      <button key={p.plugin} className="button" onClick={() => setChosen(p.plugin)}>
        {p.plugin}
      </button>
    ));
  else body = <ClaimForm plugin={plugin.plugin} form={plugin.form} />;

  return (
    <main className="login">
      <h1>Set up Innkeeper</h1>
      {body}
    </main>
  );
}

function ClaimForm({ plugin, form }: { plugin: string; form: unknown }) {
  const navigate = useNavigate();
  const [error, setError] = useState("");
  const fields = formFields(form);

  async function submit(e: FormEvent<HTMLFormElement>) {
    e.preventDefault();
    const data = new FormData(e.currentTarget);
    const params = Object.fromEntries(fields.map((f) => [f.name, String(data.get(f.name) ?? "")]));
    const res = await api.POST("/setup", { body: { code: String(data.get("code") ?? ""), plugin, params } });
    if (!res.response.ok) {
      setError((res.error as { detail?: string } | undefined)?.detail ?? res.response.statusText);
      return;
    }
    navigate("/", { replace: true });
  }

  return (
    <form onSubmit={submit} className="login-form">
      <p>Enter the setup code from the server log, then choose your password.</p>
      <label>
        Setup code
        <input className="input" name="code" required autoComplete="off" />
      </label>
      {fields.map((f) => (
        <label key={f.name}>
          {f.label}
          <input
            className="input"
            name={f.name}
            type={f.type}
            required={f.required}
            autoComplete={f.type === "password" ? "new-password" : f.autoComplete}
          />
        </label>
      ))}
      {error && <p className="error">{error}</p>}
      <button className="button primary" type="submit">
        Claim this server
      </button>
      <p className="hint">
        Or set the password in the deployment with <code>INNKEEPER_OWNER_PASSWORD_HASH</code> and restart. Other login
        providers, such as OIDC, can be added in Settings after setup.
      </p>
    </form>
  );
}
```

```go
// setupComplete checks a new password and returns its hash as the secret.
func setupComplete(params json.RawMessage) (any, error) {
	var p protocol.AuthSetupCompleteParams
	if err := json.Unmarshal(params, &p); err != nil {
		return nil, protocol.NewError(protocol.CodeInvalidParams, err.Error(), false)
	}
	pw := p.Params["password"]
	switch {
	case pw != p.Params["confirm"]:
		return nil, protocol.NewError(-32000, "the passwords don't match", false)
	case len([]rune(pw)) < minPassword:
		return nil, protocol.NewError(-32000, fmt.Sprintf("use at least %d characters", minPassword), false)
	}
	h, err := bcrypt.GenerateFromPassword([]byte(pw), cost)
	if err != nil {
		return nil, protocol.NewError(-32000, err.Error(), false)
	}
	return protocol.AuthSetupCompleteResult{Secrets: map[string]string{"password_hash": string(h)}, Subject: subject}, nil
}
```
