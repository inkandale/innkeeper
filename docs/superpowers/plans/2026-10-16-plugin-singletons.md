# Plugin singletons and the Plugins page implementation plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make one instance per plugin the default, create it automatically when the plugin needs no setup, and rebuild Settings › Plugins as grouped cards showing each plugin's status.

**Architecture:** Three new manifest fields (`multiple_instances`, `description`, `category`). `Registry.CreateInstance` enforces singletons under an advisory lock, and `Registry.EnsureSingletons` replaces `CreateBuiltinSites`. `GET /plugins` returns a computed `status`. The web page groups plugins by `category` and moves the less common actions into a popover menu.

**Tech Stack:** Go 1.25, pgx, huma; React 19, TanStack Query, Vitest, Testing Library.

// EnsureSingletons creates the instance of each loaded singleton plugin that
// needs no setup, so it works as soon as it's installed. Themes have no
// instance, and a login instance decides who signs in, so neither is made.
// A setting records each plugin it ran for, so an instance the owner
// deletes stays deleted. It returns the plugins it created instances for.
func (r *Registry) EnsureSingletons(ctx context.Context) ([]string, error) {
	manifests := r.Manifests()
	var created []string
	for _, name := range slices.Sorted(maps.Keys(manifests)) {
		m := manifests[name]
		required, _ := m.Config["required"].([]any)
		if m.MultipleInstances || m.IsTheme() || m.Has("auth") || len(m.Secrets) > 0 || len(required) > 0 {
			continue
		}
		ok, err := r.createSingletonOnce(ctx, name)
		if err != nil {
			return created, fmt.Errorf("create %s instance: %w", name, err)
		}
		if ok {
			created = append(created, name)
		}
	}
	return created, nil
}

// createSingletonOnce runs in one transaction, so the api and worker starting
// together create one instance between them.
func (r *Registry) createSingletonOnce(ctx context.Context, plugin string) (bool, error) {
	sealed, err := r.box.Seal(nil)
	if err != nil {
		return false, err
	}
	tx, err := r.pool.Begin(ctx)
	if err != nil {
		return false, err
	}
	defer tx.Rollback(ctx)
	if err := lockPlugin(ctx, tx, plugin); err != nil {
		return false, err
	}
	// The key predates singletons; renaming it would recreate deleted instances.
	tag, err := tx.Exec(ctx, `INSERT INTO setting (key, value) VALUES ($1, 'created') ON CONFLICT DO NOTHING`,
		"builtin-instance:"+plugin)
	if err != nil || tag.RowsAffected() == 0 {
		return false, err
	}
	// An owner who already has an instance, or one by that name, needs no other.
	tag, err = tx.Exec(ctx, `INSERT INTO plugin_instance (plugin, name, secrets)
		SELECT $1, $1, $2 WHERE NOT EXISTS (SELECT 1 FROM plugin_instance WHERE plugin = $1)
		ON CONFLICT (name) DO NOTHING`, plugin, sealed)
	if err != nil {
		return false, err
	}
	return tag.RowsAffected() == 1, tx.Commit(ctx)
}
```

// Status sums up a plugin and its instances for the Plugins page: broken if
// any instance is, instances for a plugin that allows many, and otherwise
// ready or needs_setup.
func Status(m *protocol.Manifest, instances []InstanceInfo) string {
	if slices.ContainsFunc(instances, func(i InstanceInfo) bool { return i.BrokenError != nil }) {
		return "broken"
	}
	switch {
	case m.MultipleInstances:
		return "instances"
	case m.IsTheme(), len(instances) > 0:
		return "ready"
	}
	return "needs_setup"
}
```

/** byCategory groups plugins under their category, alphabetically, with Other last. */
function byCategory(list: Plugin[]): [string, Plugin[]][] {
  const groups = new Map<string, Plugin[]>();
  for (const p of list) {
    const c = p.category || "Other";
    groups.set(c, [...(groups.get(c) ?? []), p]);
  }
  return [...groups].sort(([a], [b]) => (a === "Other" ? 1 : b === "Other" ? -1 : a.localeCompare(b)));
}
