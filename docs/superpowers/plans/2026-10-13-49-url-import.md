// isZip reports whether b is a zip, perhaps with data in front of it, such as
// RisuRealm's JPEG. A PNG is never one, even if its pixels hold an end record.
func isZip(b []byte) bool {
	if bytes.HasPrefix(b, []byte("PK\x03\x04")) {
		return true
	}
	const maxEnd = 22 + 0xffff // the end record and its longest comment
	return !isPNG(b) && bytes.Contains(b[max(0, len(b)-maxEnd):], []byte("PK\x05\x06"))
}
```

`archive/zip` already reads past prepended data, so `decodeCharX` needs no change. JSON can't hold the raw bytes `\x05\x06`, so JSON cards still reach `toV3`.

// Command url-import is an import plugin. It imports one card or lorebook
// from its page on a public site, such as chub.ai.
package main

import (
	"context"
	"encoding/json"
	"errors"
	"fmt"
	"os"
	"sync/atomic"

	"github.com/davidgibbons/innkeeper/protocol"
)

const version = "0.1.0"

func main() {
	// Set by initialize; requests run on other goroutines.
	var blobTmp atomic.Pointer[string]
	var conn *protocol.Conn
	f := newFetcher()
	handler := func(ctx context.Context, method string, params json.RawMessage) (any, error) {
		switch method {
		case protocol.MethodInitialize:
			var p protocol.InitializeParams
			if err := json.Unmarshal(params, &p); err != nil {
				return nil, invalid(err)
			}
			blobTmp.Store(&p.BlobTmp)
			return protocol.InitializeResult{Protocol: protocol.Version, Name: "url-import", Version: version,
				Capabilities: []string{"import"}}, nil
		case protocol.MethodShutdown:
			return nil, nil
		case protocol.MethodImportRun:
			tmp := blobTmp.Load()
			if tmp == nil {
				return nil, errors.New("import.run before initialize")
			}
			return run(ctx, f, *tmp, conn.Notify, params)
		}
		return nil, protocol.MethodNotFound(method)
	}
	conn = protocol.NewConn(os.Stdin, os.Stdout, handler)
	if err := conn.Run(context.Background()); err != nil {
		fmt.Fprintln(os.Stderr, err)
		os.Exit(1)
	}
}

package main

import (
	"context"
	"encoding/json"
	"fmt"
	"io"
	"net/http"
	"time"

	"github.com/davidgibbons/innkeeper-plugins/internal/pluginio"
	"github.com/davidgibbons/innkeeper/protocol"
)

// maxBody caps each response. RisuRealm's PNGs with embedded assets run past 10 MiB.
const maxBody = pluginio.MaxInput

// fetcher makes the plugin's requests to sites.
type fetcher struct{ http *http.Client }

func newFetcher() *fetcher { return &fetcher{http: &http.Client{Timeout: 30 * time.Second}} }

// get fetches url. site names the site in errors. A 429, a 5xx, or a network
// error is retryable, so the core re-queues the job.
func (f *fetcher) get(ctx context.Context, site, url string) ([]byte, error) {
	req, err := http.NewRequestWithContext(ctx, http.MethodGet, url, nil)
	if err != nil {
		return nil, err
	}
	// chub.ai blocks curl/ agents, and JanitorAI blocks Go's default.
	req.Header.Set("User-Agent", "innkeeper-url-import/"+version)
	resp, err := f.http.Do(req)
	if err != nil {
		if ctx.Err() != nil {
			return nil, ctx.Err()
		}
		return nil, retryable(fmt.Errorf("%s: %w", site, err))
	}
	defer resp.Body.Close()
	switch {
	case resp.StatusCode == http.StatusNotFound:
		return nil, permanent(fmt.Errorf("not found on %s", site))
	case resp.StatusCode == http.StatusTooManyRequests || resp.StatusCode >= 500:
		return nil, retryable(fmt.Errorf("%s answered %s", site, resp.Status))
	case resp.StatusCode != http.StatusOK:
		return nil, permanent(fmt.Errorf("%s answered %s", site, resp.Status))
	}
	b, err := io.ReadAll(io.LimitReader(resp.Body, maxBody+1))
	if err != nil {
		return nil, retryable(fmt.Errorf("%s: %w", site, err))
	}
	if len(b) > maxBody {
		return nil, permanent(fmt.Errorf("%s sent over %d MiB", site, maxBody>>20))
	}
	return b, nil
}

// getJSON fetches url and decodes it into v.
func (f *fetcher) getJSON(ctx context.Context, site, url string, v any) error {
	b, err := f.get(ctx, site, url)
	if err != nil {
		return err
	}
	if err := json.Unmarshal(b, v); err != nil {
		return permanent(fmt.Errorf("%s sent unreadable JSON: %w", site, err))
	}
	return nil
}

func invalid(err error) error {
	return protocol.NewError(protocol.CodeInvalidParams, err.Error(), false)
}

func retryable(err error) error {
	return protocol.NewError(protocol.CodeInternalError, err.Error(), true)
}

func permanent(err error) error {
	return protocol.NewError(protocol.CodeInternalError, err.Error(), false)
}
```

package main

import (
	"bytes"
	"context"
	"encoding/json"
	"errors"
	"fmt"
	"log"
	"net/url"
	"slices"
	"strings"

	"github.com/davidgibbons/innkeeper-plugins/card"
	"github.com/davidgibbons/innkeeper-plugins/internal/pluginio"
	"github.com/davidgibbons/innkeeper/protocol"
)

// params are an import's settings: URL, the page of a card or lorebook.
type params struct {
	URL string `json:"url"`
}

// found is what a resolver fetched: card, bytes card.Decode reads, or
// lorebook, a CCv3 character_book.
type found struct {
	item     string // the canonical page URL
	card     []byte
	avatar   []byte // any image format; nil keeps the one card.Decode returns
	lorebook json.RawMessage
	name     string // names a lorebook that has none
}

// resolver fetches the card or lorebook on the page u.
type resolver func(ctx context.Context, f *fetcher, u *url.URL) (found, error)

// resolvers maps each lowercase host to its site's resolver.
var resolvers = map[string]resolver{
	"janitorai.com":     janitor,
	"www.janitorai.com": janitor,
}

// janitor fails: anonymous requests get every definition field as null.
func janitor(context.Context, *fetcher, *url.URL) (found, error) {
	return found{}, permanent(errors.New("JanitorAI needs a login; not supported yet"))
}

// notifyFunc sends a notification to the core.
type notifyFunc func(method string, params any) error

// run answers import.run. A run is one fetch, so it sends no checkpoint.
func run(ctx context.Context, f *fetcher, blobTmp string, notify notifyFunc, raw json.RawMessage) (any, error) {
	var p protocol.ImportRunParams
	if err := json.Unmarshal(raw, &p); err != nil {
		return nil, invalid(err)
	}
	var ps params
	dec := json.NewDecoder(bytes.NewReader(p.Params))
	dec.DisallowUnknownFields()
	if err := dec.Decode(&ps); err != nil {
		return nil, invalid(fmt.Errorf(`params must be {"url": "<page URL>"}: %w`, err))
	}
	u, err := url.Parse(strings.TrimSpace(ps.URL))
	if err != nil || (u.Scheme != "http" && u.Scheme != "https") {
		return nil, invalid(fmt.Errorf("params: url %q is not an http or https URL", ps.URL))
	}
	resolve, ok := resolvers[strings.ToLower(u.Hostname())]
	if !ok {
		return nil, invalid(fmt.Errorf("no importer for %s", u.Hostname()))
	}
	got, err := resolve(ctx, f, u)
	if err != nil {
		return nil, err
	}
	if err := emit(got, p.Run, blobTmp, notify); err != nil {
		return nil, err
	}
	if err := notify(protocol.NotifyProgress, protocol.Progress{Run: p.Run, Done: 1, Total: 1}); err != nil {
		return nil, err
	}
	return struct{}{}, nil
}

// emit sends the card or lorebook got holds. A card gets its page URL in data.source.
func emit(got found, run, blobTmp string, notify notifyFunc) error {
	if got.lorebook != nil {
		book, err := card.DecodeLorebook(got.lorebook, got.name)
		if err != nil {
			return permanent(fmt.Errorf("%s: %w", got.item, err))
		}
		return notify(protocol.NotifyEmitLorebook, protocol.EmitLorebook{Run: run, Item: got.item, Lorebook: book})
	}
	c, img, err := card.Decode(got.card)
	if err != nil {
		return permanent(fmt.Errorf("%s: %w", got.item, err))
	}
	if c, err = withSource(c, got.item); err != nil {
		return permanent(fmt.Errorf("%s: %w", got.item, err))
	}
	if got.avatar != nil {
		if img, err = card.AsPNG(got.avatar); err != nil {
			// The card matters more than its picture.
			log.Printf("%s: no avatar: %v", got.item, err)
			img = nil
		}
	}
	ev := protocol.EmitCard{Run: run, Item: got.item, Card: c}
	if img != nil {
		if ev.Avatar, err = pluginio.WriteTmp(blobTmp, "avatar-*.png", img); err != nil {
			return err
		}
	}
	return notify(protocol.NotifyEmitCard, ev)
}

// withSource appends page to the CCv3 card's data.source, which core detection reads.
func withSource(c json.RawMessage, page string) (json.RawMessage, error) {
	var top, data map[string]json.RawMessage
	if err := json.Unmarshal(c, &top); err != nil {
		return nil, err
	}
	if err := json.Unmarshal(top["data"], &data); err != nil {
		return nil, fmt.Errorf("data: %w", err)
	}
	var src []any
	if s, ok := data["source"]; ok {
		if err := json.Unmarshal(s, &src); err != nil {
			return nil, fmt.Errorf("data.source: %w", err)
		}
	}
	if slices.Contains(src, any(page)) {
		return c, nil
	}
	var err error
	if data["source"], err = json.Marshal(append(src, page)); err != nil {
		return nil, err
	}
	if top["data"], err = json.Marshal(data); err != nil {
		return nil, err
	}
	return json.Marshal(top)
}

// segment reports whether s can stand as one path segment of a site URL.
func segment(s string) bool { return s != "" && s != "." && s != ".." }

package main

import (
	"context"

	"os"
	"path/filepath"

type roundTrip func(*http.Request) (*http.Response, error)

func (f roundTrip) RoundTrip(r *http.Request) (*http.Response, error) { return f(r) }

// sites sends every request to mux. Requests keep their Host, so mux
// patterns name the real host, such as "GET api.chub.ai/api/...".
func sites(t *testing.T, mux *http.ServeMux) *fetcher {
	t.Helper()
	srv := httptest.NewServer(mux)
	t.Cleanup(srv.Close)
	target, _ := url.Parse(srv.URL)
	return &fetcher{http: &http.Client{Transport: roundTrip(func(r *http.Request) (*http.Response, error) {
		r = r.Clone(r.Context())
		r.URL.Scheme, r.URL.Host = target.Scheme, target.Host
		return http.DefaultTransport.RoundTrip(r)
	})}}
}

// importURL runs import.run for page and returns what it sent.
func importURL(t *testing.T, f *fetcher, page string) ([]any, error) {
	t.Helper()
	ps, _ := json.Marshal(params{URL: page})
	raw, _ := json.Marshal(protocol.ImportRunParams{Run: "1.1", Params: ps})
	var sent []any
	_, err := run(context.Background(), f, t.TempDir(), func(_ string, p any) error {
		sent = append(sent, p)
		return nil
	}, raw)
	return sent, err
}

// emitted returns the one card import sent, failing if it sent anything else
// but progress.
func emitted(t *testing.T, sent []any) protocol.EmitCard {
	t.Helper()
	if len(sent) != 2 {
		t.Fatalf("sent %d notifications, want a card and progress: %+v", len(sent), sent)
	}
	ev, ok := sent[0].(protocol.EmitCard)
	if !ok {
		t.Fatalf("sent %T, want EmitCard", sent[0])
	}
	return ev
}

// sources returns the card's data.source.
func sources(t *testing.T, c json.RawMessage) []string {
	t.Helper()
	var v struct {

		} `json:"data"`
	}
	if err := json.Unmarshal(c, &v); err != nil {
		t.Fatal(err)
	}
	return v.Data.Source
}

// rpcErr returns err as a protocol error, failing if it isn't one.
func rpcErr(t *testing.T, err error) *protocol.Error {
	t.Helper()
	var e *protocol.Error
	if !errors.As(err, &e) {
		t.Fatalf("err = %v, want a protocol error", err)
	}
	return e
}

package main

import (
	"context"
	"encoding/json"
	"errors"
	"net/url"
	"strings"

const chubAPI = "https://api.chub.ai"

// chub resolves chub.ai and characterhub.org pages. The card it fetches
// already carries extensions.chub, which the core's chub detector reads.
func chub(ctx context.Context, f *fetcher, u *url.URL) (found, error) {
	dir, ref, ok := chubPath(u.Path)
	if !ok {
		return found{}, invalid(errors.New("a chub.ai URL looks like https://chub.ai/characters/{user}/{slug} or https://chub.ai/lorebooks/{user}/{slug}"))
	}
	item := "https://chub.ai/" + dir + "/" + ref
	var meta struct {
		Node struct {
			ID         json.Number `json:"id"`
			Name       string      `json:"name"`
			MaxResURL  string      `json:"max_res_url"`
			Definition struct {
				EmbeddedLorebook json.RawMessage `json:"embedded_lorebook"`
			} `json:"definition"`
		} `json:"node"`
	}
	if err := f.getJSON(ctx, "chub.ai", chubAPI+"/api/"+dir+"/"+ref+"?full=true", &meta); err != nil {
		return found{}, err
	}
	if dir == "lorebooks" {
		if lb := meta.Node.Definition.EmbeddedLorebook; len(lb) == 0 || string(lb) == "null" {
			return found{}, permanent(errors.New("chub.ai sent no lorebook; its definition may be hidden"))
		}
		return found{item: item, lorebook: meta.Node.Definition.EmbeddedLorebook, name: meta.Node.Name}, nil
	}
	if meta.Node.ID == "" {
		return found{}, permanent(errors.New("chub.ai sent no card ID"))
	}
	b, err := f.get(ctx, "chub.ai", chubAPI+"/api/v4/projects/"+meta.Node.ID.String()+"/repository/files/card.json/raw")
	if err != nil {
		return found{}, err
	}
	got := found{item: item, card: b}
	if meta.Node.MaxResURL != "" {
		if got.avatar, err = f.get(ctx, "chub.ai", meta.Node.MaxResURL); err != nil {
			return found{}, err
		}
	}
	return got, nil
}

// chubPath reads /characters/{user}/{slug}, /lorebooks/{user}/{slug}, or
// /{user}/{slug}, a character. ref is {user}/{slug}, path-escaped.
func chubPath(p string) (dir, ref string, ok bool) {
	parts := strings.Split(strings.Trim(p, "/"), "/")
	switch {
	case len(parts) == 3 && (parts[0] == "characters" || parts[0] == "lorebooks"):
		dir, parts = parts[0], parts[1:]
	case len(parts) == 2 && parts[0] != "characters" && parts[0] != "lorebooks":
		dir = "characters"
	default:
		return "", "", false
	}
	if !segment(parts[0]) || !segment(parts[1]) {
		return "", "", false
	}
	return dir, url.PathEscape(parts[0]) + "/" + url.PathEscape(parts[1]), true
}

package main

import (
	"context"
	"encoding/json"
	"errors"
	"fmt"
	"net/url"
	"regexp"
	"strings"
)

const (
	risuRealm  = "https://realm.risuai.net"
	risuImages = "https://sv.risuai.xyz/resource/"
)

var uuidPattern = regexp.MustCompile(`^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$`)

// risu resolves RisuRealm pages. The card it fetches already says
// risurealm:{uuid} in data.source.
func risu(ctx context.Context, f *fetcher, u *url.URL) (found, error) {
	id, ok := strings.CutPrefix(strings.TrimSuffix(u.Path, "/"), "/character/")
	if id = strings.ToLower(id); !ok || !uuidPattern.MatchString(id) {
		return found{}, invalid(errors.New("a RisuRealm URL looks like https://realm.risuai.net/character/{uuid}"))
	}
	item := risuRealm + "/character/" + id
	var page struct {
		Nodes []*struct {
			Data []json.RawMessage `json:"data"`
		} `json:"nodes"`
	}
	if err := f.getJSON(ctx, "RisuRealm", item+"/__data.json", &page); err != nil {
		return found{}, err
	}
	for _, n := range page.Nodes {
		if n == nil {
			continue
		}
		if typ, img, ok := risuCard(n.Data); ok {
			return risuDownload(ctx, f, item, id, typ, img)
		}
	}
	return found{}, permanent(errors.New("RisuRealm sent no card data"))
}

// risuCard reads card.type and card.img from one node's devalue data.
func risuCard(data []json.RawMessage) (typ, img string, ok bool) {
	at := func(i int) json.RawMessage {
		if i < 0 || i >= len(data) {
			return nil
		}
		return data[i]
	}
	var root, c map[string]int
	if len(data) == 0 || json.Unmarshal(data[0], &root) != nil {
		return "", "", false
	}
	i, ok := root["card"]
	if !ok || json.Unmarshal(at(i), &c) != nil {
		return "", "", false
	}
	_ = json.Unmarshal(at(c["type"]), &typ)
	if j, ok := c["img"]; ok {
		_ = json.Unmarshal(at(j), &img)
	}
	return typ, img, true
}

// risuDownload fetches the card in the one format its type allows; RisuRealm
// refuses the other.
func risuDownload(ctx context.Context, f *fetcher, item, id, typ, img string) (found, error) {
	switch typ {
	case "normal":
		b, err := f.get(ctx, "RisuRealm", risuRealm+"/api/v1/download/json-v3/"+id)
		if err != nil {
			return found{}, err
		}
		got := found{item: item, card: b}
		if img != "" {
			if got.avatar, err = f.get(ctx, "RisuRealm", risuImages+url.PathEscape(img)); err != nil {
				return found{}, err
			}
		}
		return got, nil
	case "charx":
		// A JPEG with the CharX appended; card.Decode returns its icon as the avatar.
		b, err := f.get(ctx, "RisuRealm", risuRealm+"/api/v1/download/charx-v3/"+id)
		return found{item: item, card: b}, err
	}
	return found{}, permanent(fmt.Errorf("RisuRealm card type %q isn't supported", typ))
}

// pathRef validates ref as exactly <a>/<b> and returns it path-escaped, so the stored URL is clean.
func pathRef(ref string) (string, bool) {
	parts := strings.Split(ref, "/")
	if len(parts) != 2 {
		return "", false
	}
	for i, p := range parts {
		if p == "" || p == "." || p == ".." || strings.ContainsAny(p, "?#\\") || strings.IndexFunc(p, func(r rune) bool { return unicode.IsSpace(r) || unicode.IsControl(r) }) >= 0 {
			return "", false
		}
		parts[i] = url.PathEscape(p)
	}
	return parts[0] + "/" + parts[1], true
}

func chubSource(dir, ref string) (Source, bool) {
	esc, ok := pathRef(ref)
	if !ok {
		return Source{}, false
	}
	return Source{Kind: "chub", Ref: ref, URL: "https://chub.ai/" + dir + "/" + esc}, true
}

// webURL reports whether u is an http or https URL on one of hosts.
func webURL(u *url.URL, hosts ...string) bool {
	return (u.Scheme == "http" || u.Scheme == "https") &&
		slices.ContainsFunc(hosts, func(h string) bool { return strings.EqualFold(u.Hostname(), h) })
}
