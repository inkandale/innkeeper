# 36b: Setup wizard — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** After the claim, a `/welcome` wizard takes the owner through choosing plugins from the catalog, setting up an instance of each, and importing cards.

**Architecture:** Two small server changes: catalog entries carry `category` and `released`, and settings carry `onboarding_done`. Everything else is web UI over existing endpoints (`/plugin-catalog`, `/plugins/{name}/install`, `/instances`, `/instances/{id}/items`, `/imports`, `/blobs`, `/jobs/{id}`). The wizard keeps no state of its own: each step reads installed plugins, instances, and jobs. The instance form from Settings is reused.

**Tech Stack:** Go (Huma, pgx), React + TanStack Query + openapi-fetch + React Router, Vitest.

Spec: `docs/superpowers/specs/2026-10-08-onboarding-design.md`, "Slice B". Wireframes: the "Choose plugins" step is a panel titled "Plugins from the official catalog" with compact cards in a grid (name, `v<version> · <released>`, two-line description, "installed" badge), grouped under small uppercase category headings.

**Test database:** every Go run sets
`INNKEEPER_TEST_DATABASE_URL=postgres://postgres:postgres@localhost:55432/postgres`.
A run where `internal/api` or `internal/plugins` tests print `SKIP` has not passed.

**Do not** `git checkout` other branches or move HEAD; work on `inn-36b-wizard`.

## Decisions made while planning

- **Connection check:** there is no generic instance check endpoint. After saving an instance of a `pull` plugin, the wizard calls `GET /instances/{id}/items`, which reads the app. Success shows "Connected"; an error shows the message with Retry and Skip. Plugins without `pull` get no check.
- **`released`** is a quoted string (`released = "2026-10-06"`). The plugins repo's `publish` tool sets it to the publish date, so the release step fills it in; `index.toml` in the repo carries `category` only.
- **Existing servers** see the wizard once after upgrading, because `onboarding_done` starts unset. "Finish later" sets it.
- **Which plugins step 3 lists:** installed plugins in a wizard category (not `Formats`, not themes) that have no instance yet, plus the ones that do, shown as done. Derived from the server, so leaving and coming back loses nothing.

// Setting is a core setting's value, or "" when unset.
func (r *Registry) Setting(ctx context.Context, key string) (string, error) {
	var v string
	err := r.pool.QueryRow(ctx, `SELECT value FROM setting WHERE key = $1`, key).Scan(&v)
	if errors.Is(err, pgx.ErrNoRows) {
		return "", nil
	}
	return v, err
}

// SetSetting stores a core setting; "" removes it.
func (r *Registry) SetSetting(ctx context.Context, key, value string) error {
	if value == "" {
		_, err := r.pool.Exec(ctx, `DELETE FROM setting WHERE key = $1`, key)
		return err
	}
	_, err := r.pool.Exec(ctx, `INSERT INTO setting (key, value) VALUES ($1, $2)
		ON CONFLICT (key) DO UPDATE SET value = EXCLUDED.value`, key, value)
	return err
}

// Theme is the selected theme plugin's name, or "" for the default theme.
func (r *Registry) Theme(ctx context.Context) (string, error) { return r.Setting(ctx, "theme") }

// SetTheme selects an installed theme plugin, or the default theme for "".
func (r *Registry) SetTheme(ctx context.Context, name string) error {
	if name != "" {
		if m, ok := r.manifest(name); !ok || !m.IsTheme() {
			return fmt.Errorf("%w: %q is not an installed theme", ErrInvalid, name)
		}
	}
	return r.SetSetting(ctx, "theme", name)
}

```ts
import type { components } from "../api/schema";

type CatalogSource = components["schemas"]["CatalogSourceOut"];
type CatalogPlugin = components["schemas"]["CatalogPluginOut"];

export type Offer = CatalogPlugin & { source: string };
export type Group = { category: string; plugins: Offer[] };

// Categories the wizard knows, in the order it shows them.
const known = ["Apps", "Import", "Login"];

/** wizardPlugin reports whether the wizard offers a plugin: codecs are built in, themes live in Appearance. */
export function wizardPlugin(p: { capabilities?: string[] | null; category?: string | null }): boolean {
  const caps = p.capabilities ?? [];
  return !caps.includes("codec") && !caps.includes("theme") && p.category !== "Formats";
}

/** groupCatalog groups the plugins every source offers by category; the first source to offer a name wins. */
export function groupCatalog(sources: CatalogSource[]): Group[] {
  const seen = new Set<string>();
  const byCategory = new Map<string, Offer[]>();
  for (const s of sources) {
    for (const p of s.plugins ?? []) {
      if (seen.has(p.name) || !wizardPlugin(p)) continue;
      seen.add(p.name);
      const c = p.category || "Other";
      byCategory.set(c, [...(byCategory.get(c) ?? []), { ...p, source: s.source.id }]);
    }
  }
  const rank = (c: string) => (known.includes(c) ? known.indexOf(c) : c === "Other" ? Infinity : known.length);
  return [...byCategory]
    .sort(([a], [b]) => rank(a) - rank(b) || a.localeCompare(b))
    .map(([category, plugins]) => ({ category, plugins }));
}
```

Use the generated schema names; if the catalog plugin schema is named differently than `CatalogPluginOut`, use that name.

/** useFinish marks the wizard done and goes to the library. */
export function useFinish() {
  const qc = useQueryClient();
  const navigate = useNavigate();
  return useMutation({
    mutationFn: async () => must(await api.PATCH("/settings", { body: { onboarding_done: true } })),
    onSuccess: (s) => {
      qc.setQueryData(["settings"], s);
      navigate("/cards", { replace: true });

/** Welcome is the setup wizard: a rail of steps beside the current one. */
