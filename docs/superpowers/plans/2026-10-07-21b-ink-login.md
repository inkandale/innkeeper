# ink login Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** `ink login <url>` signs in through the browser and keeps the login in the OS keychain, `ink logout` revokes it, and named profiles cover more than one server.

**Architecture:** `ink` uses the OAuth flow MCP clients already use: dynamic registration, authorization code with PKCE, and a loopback redirect. The server gains three things for it. Each grant is bound to a resource, `/mcp` or `/api`, and a token works only at its own. A loopback redirect may use any port (RFC 8252 §7.3). `POST /oauth/revoke` revokes a token (RFC 7009). On the client, profiles live in a JSON config file, and each profile's tokens live in the keychain, or in a file only the user can read where there is no keychain. `INNKEEPER_URL` and `INNKEEPER_TOKEN` still take precedence, for scripts.

**Tech Stack:** Go 1.27, `github.com/spf13/cobra`, `github.com/zalando/go-keyring`, PostgreSQL.

Ticket: INN-21, slice 21b. The slices are listed in `docs/superpowers/plans/2026-10-07-21a-ink-client.md`.

Decisions recorded during planning:

- A grant's resource defaults to `mcp` when the authorization request names none, so MCP clients that send no `resource` keep working. Existing grants become `mcp` grants.
- The API takes an `api` access token on any request, and an `mcp` one only on requests the MCP server makes in process, as today. `/mcp` takes only `mcp` access tokens. API tokens work in both places, as today.
- Revoking either token of a grant deletes the grant, which ends both tokens. The token must belong to the `client_id` the request names; an unknown token is not an error (RFC 7009 §2.2).
- `ink` registers a new client at each login, with the portless redirect `http://127.0.0.1/callback`, and keeps its ID in the profile for refreshes. Clients left with no grant are pruned after a day, as today.
- Logging in again to a profile revokes its old login first, so replaced logins don't stay live.
- `ink` stores the access token, its expiry, and the refresh token together, and refreshes only within a minute of expiry. If a refresh is refused, `ink` rereads the store once, since another `ink` process may have just refreshed.
- Profile names match `^[A-Za-z0-9_-]{1,32}$`, since a name becomes part of a file path.
- No OAuth access token, whatever its resource or tier, can change login providers or owners, so an `ink` login can't make itself the owner any more than an MCP client can. API tokens and browser sessions still can.
- Grants keep their resource to themselves: the grant list and the web UI don't show it, since the client name already says what the client is.

Not in this slice: `/.well-known/oauth-protected-resource/api` and a `WWW-Authenticate` challenge on `/api`. `ink` doesn't need them; add them when another client discovers the API through OAuth.

```go
// register registers a client that redirects to redirectURI, and returns
// its ID.
func (o *oauthEnv) register(redirectURI string) string {
	o.t.Helper()
	resp, err := http.Post(o.url+"/oauth/register", "application/json",
		strings.NewReader(`{"client_name": "ink", "redirect_uris": ["`+redirectURI+`"]}`))
	if err != nil {
		o.t.Fatal(err)
	}
	defer resp.Body.Close()
	var c struct {
		ID string `json:"client_id"`
	}
	if err := json.NewDecoder(resp.Body).Decode(&c); err != nil || c.ID == "" {
		o.t.Fatalf("register: %d, %v", resp.StatusCode, err)
	}
	return c.ID

// authParams returns a PKCE verifier and an authorization request from
// clientID for resource, or for no resource if it is "".
func authParams(clientID, redirectURI, resource string) (string, url.Values) {
	verifier := strings.Repeat("x", 50)
	sum := sha256.Sum256([]byte(verifier))
	p := url.Values{"response_type": {"code"}, "client_id": {clientID}, "redirect_uri": {redirectURI},
		"code_challenge": {base64.RawURLEncoding.EncodeToString(sum[:])}, "code_challenge_method": {"S256"},
		"state": {"xyz"}}
	if resource != "" {
		p.Set("resource", resource)
	}
	return verifier, p
}

// approve logs in if it must, approves params at tier, and returns the code
// sent back, or "" if none was.
func (o *oauthEnv) approve(params url.Values, tier string) string {
	o.t.Helper()
	_, page := o.get("/oauth/authorize?" + params.Encode())
	if strings.Contains(page, `name="f_password"`) {
		o.post("/oauth/login", url.Values{"provider": {itoa64(o.loginID)}, "return": {params.Encode()}, "f_password": {"letmein"}})
		_, page = o.get("/oauth/authorize?" + params.Encode())
	}
	m := csrfField.FindStringSubmatch(page)
	if m == nil {
		o.t.Fatalf("consent page = %s", page)
	}
	decision := url.Values{"csrf": {m[1]}, "tier": {tier}, "decision": {"approve"}}
	for k, v := range params {
		decision[k] = v
	}
	resp, _ := o.post("/oauth/authorize", decision)
	back, _ := url.Parse(resp.Header.Get("Location"))
	return back.Query().Get("code")
}

// redeem exchanges code for an access token and a refresh token.
func (o *oauthEnv) redeem(clientID, code, redirectURI, verifier string) (string, string) {
	o.t.Helper()
	_, body := o.post("/oauth/token", url.Values{"grant_type": {"authorization_code"}, "code": {code},
		"client_id": {clientID}, "redirect_uri": {redirectURI}, "code_verifier": {verifier}})
	var toks struct {
		Access  string `json:"access_token"`
		Refresh string `json:"refresh_token"`
	}
	if json.Unmarshal([]byte(body), &toks) != nil || toks.Access == "" {
		o.t.Fatalf("token = %s", body)
	}
	return toks.Access, toks.Refresh
}

// bearerStatus sends method path with token and returns the status.
func (o *oauthEnv) bearerStatus(method, path, token string) int {
	o.t.Helper()
	req, _ := http.NewRequest(method, o.url+path, strings.NewReader("{}"))
	req.Header.Set("Authorization", "Bearer "+token)
	req.Header.Set("Content-Type", "application/json")
	resp, err := http.DefaultClient.Do(req)
	if err != nil {
		o.t.Fatal(err)
	}
	resp.Body.Close()
	return resp.StatusCode
}
```

```go
// validRedirect accepts https URIs, and http ones on a loopback host.
func validRedirect(raw string) bool {
	if len(raw) > 2048 {
		return false
	}
	u, err := url.Parse(raw)
	if err != nil || u.Fragment != "" || u.Host == "" {
		return false
	}
	return u.Scheme == "https" || u.Scheme == "http" && loopback(u.Hostname())
}

func loopback(host string) bool { return host == "localhost" || host == "127.0.0.1" || host == "::1" }

// redirectAllowed reports whether uri is one of registered. A loopback
// redirect may use any port, since a native app learns its port only when
// it starts listening (RFC 8252 §7.3).
func redirectAllowed(registered []string, uri string) bool {
	if slices.Contains(registered, uri) {
		return true
	}
	u, err := url.Parse(uri)
	if err != nil || u.Scheme != "http" || !loopback(u.Hostname()) {
		return false
	}
	return slices.ContainsFunc(registered, func(r string) bool {
		ru, err := url.Parse(r)
		return err == nil && ru.Scheme == "http" && ru.Hostname() == u.Hostname() && ru.Path == u.Path && ru.RawQuery == u.RawQuery
	})
}
```

`RedeemCode` still compares the token request's `redirect_uri` with the one the code was issued for, port included, as RFC 6749 §4.1.3 requires.

```go
// RevokeOAuthToken ends the grant that issued token, if clientID holds it.
// An unknown token is not an error (RFC 7009 §2.2).
func (a *Auth) RevokeOAuthToken(ctx context.Context, clientID, token string) error {
	_, err := a.pool.Exec(ctx, `DELETE FROM oauth_grant WHERE client_id = $1
		AND id = (SELECT grant_id FROM oauth_token WHERE hash = $2)`, clientID, hash(token))
	return err
}
```

```go
func (s *Server) revoke(w http.ResponseWriter, r *http.Request) {
	r.Body = http.MaxBytesReader(w, r.Body, 64<<10)
	if err := r.ParseForm(); err != nil || r.PostForm.Get("token") == "" {
		oauthError(w, http.StatusBadRequest, "invalid_request", "the body must be form-encoded and name a token")
		return

	if r.PostForm.Get("client_id") == "" {
		oauthError(w, http.StatusUnauthorized, "invalid_client", "client_id is required")
```

```go
package main

import (
	"encoding/json"
	"errors"
	"io/fs"
	"os"
	"path/filepath"
	"time"

	"github.com/zalando/go-keyring"
)

// creds are a profile's OAuth tokens.
type creds struct {
	Access  string    `json:"access"`
	Expires time.Time `json:"expires"`
	Refresh string    `json:"refresh"`
}

const keyringService = "ink"

var errNoCreds = errors.New("no saved login")

// credsFile is where a profile's tokens go when there is no keychain.
func credsFile(name string) (string, error) {
	if err := checkProfileName(name); err != nil {
		return "", err
	}
	dir, err := configDir()
	return filepath.Join(dir, name+".creds"), err
}

// saveCreds stores a profile's tokens in the OS keychain or, where there is
// none, in a file only the user can read. It returns which.
func saveCreds(name string, c creds) (string, error) {
	path, err := credsFile(name)
	if err != nil {
		return "", err
	}
	b, err := json.Marshal(c)
	if err != nil {
		return "", err
	}
	if keyring.Set(keyringService, name, string(b)) == nil {
		if err := os.Remove(path); err != nil && !errors.Is(err, fs.ErrNotExist) {
			return "", err
		}
		return "the keychain", nil
	}
	if err := os.MkdirAll(filepath.Dir(path), 0o700); err != nil {
		return "", err
	}
	return path, os.WriteFile(path, b, 0o600)
}

// loadCreds returns a profile's tokens. A file means the keychain failed
// when they were saved, so it comes first.
func loadCreds(name string) (creds, error) {
	var c creds
	path, err := credsFile(name)
	if err != nil {
		return c, err
	}
	b, err := os.ReadFile(path)
	if errors.Is(err, fs.ErrNotExist) {
		s, kerr := keyring.Get(keyringService, name)
		if errors.Is(kerr, keyring.ErrNotFound) {
			return c, errNoCreds
		}
		if kerr != nil {
			return c, kerr
		}
		b, err = []byte(s), nil
	}
	if err != nil {
		return c, err
	}
	return c, json.Unmarshal(b, &c)
}

func deleteCreds(name string) error {
	path, err := credsFile(name)
	if err != nil {
		return err
	}
	if err := os.Remove(path); !errors.Is(err, fs.ErrNotExist) {
		return err
	}
	if err := keyring.Delete(keyringService, name); err != nil && !errors.Is(err, keyring.ErrNotFound) {
		return err
	}
	return nil
}
```

```go
package main

import (
	"cmp"
	"context"
	"encoding/json"
	"errors"
	"fmt"
	"io"
	"net/http"
	"net/url"
	"strings"
	"time"
)

// tokenResponse is a token endpoint's reply.
type tokenResponse struct {
	Access    string `json:"access_token"`
	Refresh   string `json:"refresh_token"`
	ExpiresIn int    `json:"expires_in"`
	Scope     string `json:"scope"`
}

func (t tokenResponse) creds() creds {
	return creds{Access: t.Access, Refresh: t.Refresh, Expires: time.Now().Add(time.Duration(t.ExpiresIn) * time.Second)}
}

// oauthError is an error an OAuth endpoint returned.
type oauthError struct {
	Status      int
	Code        string `json:"error"`
	Description string `json:"error_description"`
}

func (e *oauthError) Error() string { return cmp.Or(e.Description, e.Code, http.StatusText(e.Status)) }

// postForm posts form to endpoint and decodes the JSON reply into out,
// unless out is nil.
func postForm(ctx context.Context, endpoint string, form url.Values, out any) error {
	req, err := http.NewRequestWithContext(ctx, http.MethodPost, endpoint, strings.NewReader(form.Encode()))
	if err != nil {
		return err
	}
	req.Header.Set("Content-Type", "application/x-www-form-urlencoded")
	resp, err := http.DefaultClient.Do(req)
	if err != nil {
		return err
	}
	defer resp.Body.Close()
	raw, err := io.ReadAll(io.LimitReader(resp.Body, 1<<20))
	if err != nil {
		return err
	}
	if resp.StatusCode >= 300 {
		e := &oauthError{Status: resp.StatusCode}
		_ = json.Unmarshal(raw, e)
		return e
	}
	if out == nil {
		return nil
	}
	return json.Unmarshal(raw, out)
}

// accessToken returns the profile's access token, refreshing it within a
// minute of its expiry.
func accessToken(ctx context.Context, name string, p profile) (string, error) {
	c, err := loadCreds(name)
	if errors.Is(err, errNoCreds) {
		return "", fmt.Errorf("profile %s has no saved login; log in with: ink login --profile %s %s", name, name, p.URL)
	}
	if err != nil {
		return "", err
	}
	if time.Until(c.Expires) > time.Minute {
		return c.Access, nil
	}
	var toks tokenResponse
	err = postForm(ctx, p.URL+"/oauth/token", url.Values{"grant_type": {"refresh_token"},
		"refresh_token": {c.Refresh}, "client_id": {p.ClientID}}, &toks)
	if oe := (*oauthError)(nil); errors.As(err, &oe) && oe.Code == "invalid_grant" {
		// Another ink may have just refreshed; it saved what it got.
		if again, err := loadCreds(name); err == nil && again.Refresh != c.Refresh && time.Until(again.Expires) > time.Minute {
			return again.Access, nil
		}
		return "", fmt.Errorf("the login for profile %s has ended; log in again with: ink login --profile %s %s", name, name, p.URL)
	}
	if err != nil {
		return "", fmt.Errorf("refresh the login for profile %s: %w", name, err)
	}
	if _, err := saveCreds(name, toks.creds()); err != nil {
		return "", err
	}
	return toks.Access, nil
}
```

```go
// login runs the OAuth flow against server and returns the profile and the
// tokens it got.
func login(ctx context.Context, stderr io.Writer, server string) (profile, tokenResponse, error) {
	var meta struct {
		Issuer     string `json:"issuer"`
		Revocation string `json:"revocation_endpoint"`
	}
	if err := getJSON(ctx, server+"/.well-known/oauth-authorization-server", &meta); err != nil || meta.Issuer == "" {
		return profile{}, tokenResponse{}, fmt.Errorf("%s doesn't offer login (%v); set INNKEEPER_PUBLIC_URL on the server, or use INNKEEPER_URL and INNKEEPER_TOKEN", server, err)
	}
	if meta.Revocation == "" {
		return profile{}, tokenResponse{}, fmt.Errorf("%s is too old for ink login; upgrade it", server)
	}
	issuer := meta.Issuer
	ln, err := net.Listen("tcp", "127.0.0.1:0")
	if err != nil {
		return profile{}, tokenResponse{}, err
	}
	defer ln.Close()
	redirect := fmt.Sprintf("http://%s/callback", ln.Addr())
	clientID, err := register(ctx, issuer)
	if err != nil {
		return profile{}, tokenResponse{}, err
	}
	verifier, state := randomString(32), randomString(16)
	sum := sha256.Sum256([]byte(verifier))
	page := issuer + "/oauth/authorize?" + url.Values{"response_type": {"code"}, "client_id": {clientID},
		"redirect_uri": {redirect}, "code_challenge": {base64.RawURLEncoding.EncodeToString(sum[:])},
		"code_challenge_method": {"S256"}, "state": {state}, "resource": {issuer + "/api"}}.Encode()
	fmt.Fprintf(stderr, "Opening %s\nIf no browser opens, visit that address.\n", page)
	_ = openBrowser(page)
	code, err := awaitCode(ctx, ln, state, issuer)
	if err != nil {
		return profile{}, tokenResponse{}, err
	}
	var toks tokenResponse
	err = postForm(ctx, issuer+"/oauth/token", url.Values{"grant_type": {"authorization_code"}, "code": {code},
		"client_id": {clientID}, "redirect_uri": {redirect}, "code_verifier": {verifier}}, &toks)
	return profile{URL: issuer, ClientID: clientID}, toks, err
}

func randomString(n int) string {
	b := make([]byte, n)
	rand.Read(b)
	return base64.RawURLEncoding.EncodeToString(b)
}

func getJSON(ctx context.Context, u string, out any) error {
	req, err := http.NewRequestWithContext(ctx, http.MethodGet, u, nil)
	if err != nil {
		return err
	}
	resp, err := http.DefaultClient.Do(req)
	if err != nil {
		return err
	}
	defer resp.Body.Close()
	if resp.StatusCode != http.StatusOK {
		return errors.New(resp.Status)
	}
	return json.NewDecoder(io.LimitReader(resp.Body, 1<<20)).Decode(out)
}

// register registers ink as an OAuth client of issuer and returns its ID.
// The redirect has no port, so the server accepts whichever one ink gets.
func register(ctx context.Context, issuer string) (string, error) {
	host, _ := os.Hostname()
	body, _ := json.Marshal(map[string]any{"client_name": "ink on " + cmp.Or(host, "this computer"),
		"redirect_uris": []string{"http://127.0.0.1/callback"}})
	req, err := http.NewRequestWithContext(ctx, http.MethodPost, issuer+"/oauth/register", bytes.NewReader(body))
	if err != nil {
		return "", err
	}
	req.Header.Set("Content-Type", "application/json")
	resp, err := http.DefaultClient.Do(req)
	if err != nil {
		return "", err
	}
	defer resp.Body.Close()
	var out struct {
		ClientID    string `json:"client_id"`
		Description string `json:"error_description"`
	}
	_ = json.NewDecoder(io.LimitReader(resp.Body, 1<<20)).Decode(&out)
	if resp.StatusCode != http.StatusCreated || out.ClientID == "" {
		return "", fmt.Errorf("register with %s: %s", issuer, cmp.Or(out.Description, resp.Status))
	}
	return out.ClientID, nil
}

// awaitCode serves the redirect on ln and returns the code the browser
// brings back.
func awaitCode(ctx context.Context, ln net.Listener, state, issuer string) (string, error) {
	type result struct {
		code string
		err  error
	}
	done := make(chan result, 1)
	srv := &http.Server{ReadHeaderTimeout: 10 * time.Second, Handler: http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		if r.URL.Path != "/callback" {
			http.NotFound(w, r)
			return
		}
		q := r.URL.Query()
		res := result{code: q.Get("code")}
		switch {
		case q.Get("state") != state:
			res.err = errors.New("the login came back with the wrong state; start it again")
		case q.Get("iss") != issuer:
			res.err = fmt.Errorf("the login came back from %q, not %s", q.Get("iss"), issuer)
		case q.Get("error") != "":
			res.err = fmt.Errorf("login refused: %s", cmp.Or(q.Get("error_description"), q.Get("error")))
		case res.code == "":
			res.err = errors.New("the login came back without a code")
		}
		if res.err != nil {
			http.Error(w, res.err.Error(), http.StatusBadRequest)
		} else {
			fmt.Fprintln(w, "ink is logged in. You can close this tab.")
		}
		select {
		case done <- res:
		default:
		}
	})}
	go srv.Serve(ln)
	defer srv.Close()
	ctx, cancel := context.WithTimeout(ctx, 5*time.Minute)
	defer cancel()
	select {
	case r := <-done:
		return r.code, r.err
	case <-ctx.Done():
		return "", errors.New("timed out waiting for the login in the browser")
	}
}
```
