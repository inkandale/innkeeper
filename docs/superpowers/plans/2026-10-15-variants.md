# Variants Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A card or lorebook can record the entity it came from, and the library list folds each family of variants into one row.

**Architecture:** INN-52, spec `docs/superpowers/specs/2026-10-15-variants-design.md`. One plan in three slices, each mergeable on its own:

1. **Model** (Tasks 1–8): `variant_of_id` on `card` and `lorebook`, `Entities.SetVariantOf` with a cycle check, `Entities.Fork`, the routes, two MCP tools, the card and lorebook pages, and `CardPicker`.
2. **Grouped search** (Tasks 9–12): `group: "family"` and the `variant_of_root` filter in `library.Search`, and collapsed family rows in the library list.
3. **Reconcile and bulk** (Tasks 13–18): "Import as variant of this", the `shares_source` filter and its toggle, and bulk "Make variants of…".

Everything goes through `library.Kind`, so cards and lorebooks share one implementation. The only kind-specific part is what a fork copies besides the revision: a card's collections and lorebook links. That list lives on the kind (`Kind.forkCopies`), not in a `switch`.

**Tech Stack:** Go, pgx, PostgreSQL, Huma, MCP Go SDK, React, TypeScript, TanStack Query, Vitest with Testing Library.

**Tests:** DB tests skip without the database URL, so a run without it proves nothing. Run every Go test with
`INNKEEPER_TEST_DATABASE_URL=postgres://postgres:postgres@localhost:55432/postgres` and `-timeout 15m`. Below, `$DB` means that assignment. Run web tests from `web/` with `npx vitest run <file>`. The API schema is generated: after a Go change to a request or response, run `npm run gen` in `web/`, and never edit `web/openapi.json` or `web/src/api/schema.d.ts` by hand. No commit may be marked as a breaking change.

**Deviations from the spec**, forced by the code:
- `SetVariantOf` and `Fork` are methods on `*Entities`, which already holds its kind, so they take no `k` argument. `Fork` also takes the revision `source` (`ui` or `mcp`), because `card_revision.source` must be one of a fixed set.
- The foreign keys are `DEFERRABLE INITIALLY IMMEDIATE`: `TestForeignKeysAreDeferrable` requires it, since restore defers foreign key checks.
- Huma can't mark a struct field nullable, so GET leaves `variant_of` out when there is no parent instead of sending `null`.
- A fork's `provenance` is `variant`. The spec doesn't set it, and the source's provenance would let an import of the source match the fork's content.
- "The oldest" selected entity in bulk "Make variants of…" is the lowest ID, since IDs are assigned in creation order. The action needs the picked IDs, so it isn't offered when "Select all matching" is on.

```go
package library

import (
	"context"
	"fmt"

	"github.com/jackc/pgx/v5"
)

// EntityRef names a card or lorebook by its ID, slug, and head revision's name.
type EntityRef struct {
	ID   int64  `json:"id"`
	Slug string `json:"slug"`
	Name string `json:"name"`
}

// SetVariantOf makes parent the entity's parent, or clears the link when
// parent is nil. The link isn't content, so it makes no revision.
func (s *Entities) SetVariantOf(ctx context.Context, id int64, parent *int64) error {
	return pgx.BeginFunc(ctx, s.pool, func(tx pgx.Tx) error { return s.SetVariantOfTx(ctx, tx, id, parent) })
}

// SetVariantOfTx is SetVariantOf inside the caller's transaction. A parent
// that is id or a variant below it is ErrVariantCycle; a parent this kind
// doesn't have is ErrNotFound.
func (s *Entities) SetVariantOfTx(ctx context.Context, tx pgx.Tx, id int64, parent *int64) error {
	if parent != nil {
		// Two links checked at once could each pass and together make a cycle.
		if _, err := tx.Exec(ctx, `SELECT pg_advisory_xact_lock(hashtext($1))`, s.k.Table+":variant_of"); err != nil {
			return err
		}
		var found int
		var cycle bool
		err := tx.QueryRow(ctx, `WITH RECURSIVE up AS (
				SELECT id, variant_of_id FROM `+s.k.Table+` WHERE id = $1
				UNION
				SELECT p.id, p.variant_of_id FROM `+s.k.Table+` p JOIN up ON p.id = up.variant_of_id)
			SELECT count(*), coalesce(bool_or(id = $2), false) FROM up`, *parent, id).Scan(&found, &cycle)
		switch {
		case err != nil:
			return err
		case found == 0:
			return fmt.Errorf("%w: %s %d", ErrNotFound, s.k.Name, *parent)
		case cycle:
			return ErrVariantCycle
		}
	}
	tag, err := tx.Exec(ctx, `UPDATE `+s.k.Table+` SET variant_of_id = $2 WHERE id = $1`, id, parent)
	if err != nil {
		return mapErr(err)
	}
	if tag.RowsAffected() == 0 {
		return ErrNotFound
	}
	return nil
}

// Parent returns the entity this one is a variant of, or nil.
func (s *Entities) Parent(ctx context.Context, id int64) (*EntityRef, error) {
	refs, err := s.refs(ctx, `c.id = (SELECT variant_of_id FROM `+s.k.Table+` WHERE id = $1)`, id)
	if err != nil || len(refs) == 0 {
		return nil, err
	}
	return &refs[0], nil
}

// Variants returns the entity's direct variants, oldest first.
func (s *Entities) Variants(ctx context.Context, id int64) ([]EntityRef, error) {
	return s.refs(ctx, `c.variant_of_id = $1`, id)
}

// refs returns the entities matching cond, whose one argument is id.
func (s *Entities) refs(ctx context.Context, cond string, id int64) ([]EntityRef, error) {
	rows, err := s.pool.Query(ctx, `SELECT c.id, c.slug, coalesce(r.data->>'name', '') FROM `+from(s.k)+`
		WHERE `+cond+` ORDER BY c.id`, id)
	if err != nil {
		return nil, err
	}
	return pgx.CollectRows(rows, pgx.RowToStructByPos[EntityRef])
}
```

The `UNION` (not `UNION ALL`) in the walk up stops at a repeated row, so even a cycle written by hand can't loop forever.

```go
// Fork creates a variant of entity id from its head revision: its data,
// avatar, and overrides, and the kind's links, but not its sources or
// bindings. A shared source is reconcile's strongest match signal, so
// copying one would make the two hard to tell apart. source is the new
// revision's source, ui or mcp.
func (s *Entities) Fork(ctx context.Context, id int64, source string) (Entity, error) {
	var ent Entity
	err := pgx.BeginFunc(ctx, s.pool, func(tx pgx.Tx) error {
		from, err := scanEntity(tx.QueryRow(ctx, `SELECT `+entityCols+` FROM `+s.k.Table+` WHERE id = $1`, id))
		if err != nil {
			return err
		}
		head, err := scanRevision(tx.QueryRow(ctx, `SELECT `+s.revCols()+` FROM `+s.k.RevTable+` WHERE id = $1`, from.HeadRevisionID))
		if err != nil {
			return err
		}
		e := Edit{Data: head.Data, Overrides: head.Overrides, Source: source,
			Message: fmt.Sprintf("Variant of %s at r%d", from.Slug, head.ID)}
		if head.AvatarBlob != nil {
			e.AvatarBlob = *head.AvatarBlob
		}
		r, err := s.prepare(e)
		if err != nil {
			return err
		}
		if ent, _, err = s.createFree(ctx, tx, from.Slug+"-variant", "variant", r); err != nil {
			return err
		}
		for _, q := range s.k.forkCopies {
			if _, err := tx.Exec(ctx, q, id, ent.ID); err != nil {
				return err
			}
		}
		return s.SetVariantOfTx(ctx, tx, ent.ID, &id)
	})
	return ent, err
}
```

```tsx
  );
```

```go
// familiesOf is a WITH clause whose fam holds each family with a match for
// cond: its root, its best match's place under order, and its matches.
func familiesOf(k *Kind, cond, order string) string {
	return `WITH RECURSIVE m AS (
		SELECT c.id, row_number() OVER (ORDER BY ` + order + `) AS pos FROM ` + from(k) + ` WHERE ` + cond + `
	), up AS (
		SELECT m.id AS member, e.id, e.variant_of_id FROM m JOIN ` + k.Table + ` e ON e.id = m.id
		UNION ALL
		SELECT up.member, p.id, p.variant_of_id FROM up JOIN ` + k.Table + ` p ON p.id = up.variant_of_id
	), fam AS (
		SELECT up.id AS root_id, min(m.pos) AS pos, array_agg(m.id ORDER BY m.id) AS matched
		FROM up JOIN m ON m.id = up.member WHERE up.variant_of_id IS NULL GROUP BY up.id
	)`
}

// familyPage follows familiesOf: it selects one page of fam, best placed
// first, with each family's face and size. The face is the shallowest live
// member, then the oldest, then the lowest ID; with no live member, the root.
// all counts archived members in size.
func familyPage(k *Kind, all bool, limit, offset string) string {
	size := `count(*) FILTER (WHERE e.archived_at IS NULL)::int`
	if all {
		size = `count(*)::int`
	}
	return `, pg AS (
		SELECT * FROM fam ORDER BY pos LIMIT ` + limit + ` OFFSET ` + offset + `
	), down AS (
		SELECT pg.root_id, pg.root_id AS id, 0 AS depth FROM pg
		UNION ALL
		SELECT down.root_id, e.id, down.depth + 1 FROM down JOIN ` + k.Table + ` e ON e.variant_of_id = down.id
	)
	SELECT pg.root_id, face.id, face.size, pg.matched FROM pg CROSS JOIN LATERAL (
		SELECT (array_agg(e.id ORDER BY e.archived_at IS NOT NULL, d.depth, e.created_at, e.id))[1] AS id, ` + size + ` AS size
		FROM down d JOIN ` + k.Table + ` e ON e.id = d.id WHERE d.root_id = pg.root_id
	) face ORDER BY pg.pos`
}
```

```go
	rows, err := l.pool.Query(ctx, sql, args...)
	if err != nil {
		return nil, err
	}
	out := []Hit{}
	var h Hit
	var data []byte
	var avatar *string
	_, err = pgx.ForEachRow(rows, []any{&h.ID, &h.Slug, &h.RevisionID, &h.ArchivedAt, &data, &avatar, &h.Tags, &h.Lorebooks, &h.Targets, &h.Cards, &h.Entries},
		func() error {
			var all map[string]any
			if err := json.Unmarshal(data, &all); err != nil {
				return err
			}
			h.Fields = map[string]any{}
			for _, f := range fields {

					// The avatar is the blob only, never a same-named data key.

					}

// faces runs page, a familiesOf and familyPage query, and returns each
// family's face as a hit with its family, in page order. cols selects the
// hit columns from the kind's entities.
func (l *Library) faces(ctx context.Context, page string, args []any, cols string, fields []string) ([]Hit, error) {
	rows, err := l.pool.Query(ctx, page, args...)
	if err != nil {
		return nil, err
	}
	type face struct {
		id     int64
		family Family
	}
	var list []face
	var f face
	if _, err := pgx.ForEachRow(rows, []any{&f.family.RootID, &f.id, &f.family.Size, &f.family.Matched}, func() error {
		list = append(list, f)
		f = face{}
		return nil
	}); err != nil {
		return nil, err
	}
	ids := make([]int64, len(list))
	for i, f := range list {
		ids[i] = f.id
	}
	found, err := l.hits(ctx, cols+` WHERE c.id = ANY($1)`, []any{ids}, fields)
	if err != nil {
		return nil, err
	}
	byID := make(map[int64]Hit, len(found))
	for _, h := range found {
		byID[h.ID] = h
	}
	out := make([]Hit, len(list))
	for i, f := range list {
		out[i] = byID[f.id]
		out[i].Family = &f.family
	}
```

```tsx
import { useMutation, useQueries, useQueryClient } from "@tanstack/react-query";
import { useState } from "react";
import { api, must, mustOk, route, type Kind } from "../api/client";
import { useToast } from "../shell/Toasts";
import { CardPicker } from "./CardPicker";

type SourceKey = { kind: string; ref: string };

/** sharedSource says whether every list is loaded and some source is in all of them. */
export function sharedSource(lists: (SourceKey[] | undefined)[]): boolean {
  if (!lists.length || lists.some((l) => !l)) return false;
  const [first, ...rest] = lists as SourceKey[][];
  return first!.some((s) => rest.every((l) => l.some((o) => o.kind === s.kind && o.ref === s.ref)));
}

/** MakeVariants makes each picked card or lorebook a variant of one parent. When they all share a source, the parent
 * starts as the oldest of them: IDs are assigned in creation order, so the lowest. */
export function MakeVariants({ kind, ids, nameOf, onDone }: { kind: Kind; ids: number[]; nameOf: Map<number, string>; onDone: () => void }) {
  const qc = useQueryClient();
  const toast = useToast();
  const [picked, setPicked] = useState<{ id: number; name: string }>();
  const sources = useQueries({
    queries: ids.map((id) => ({
      queryKey: [kind, id, "sources"],
      queryFn: async () => must(await api.GET(route(kind, "/cards/{id}/sources"), { params: { path: { id } } })),
    })),
  });
  const name = (id: number) => nameOf.get(id) || `#${id}`;
  const oldest = sharedSource(sources.map((s) => s.data)) ? Math.min(...ids) : undefined;
  const parent = picked ?? (oldest === undefined ? undefined : { id: oldest, name: name(oldest) });
  const make = useMutation({
    mutationFn: async (to: { id: number; name: string }) => {
      const failed: string[] = [];
      for (const id of ids.filter((i) => i !== to.id)) {
        try {
          mustOk(await api.PUT(route(kind, "/cards/{id}/variant_of"), { params: { path: { id } }, body: { id: to.id } }));
        } catch (e) {
          failed.push(`${name(id)}: ${(e as Error).message}`);
        }
      }
      void qc.invalidateQueries({ queryKey: ["search"] });
      if (failed.length) throw new Error(failed.join("; "));
    },
    onSuccess: (_, to) => {
      toast({ text: `Made them variants of ${to.name}.` });
      onDone();
    },
  });
  return (
    <>
      {parent && <span>Parent: {parent.name}</span>}
      <CardPicker kind={kind} label="Parent" onPick={(id, n) => setPicked({ id, name: n })} />
      <button className="button primary" disabled={!parent || make.isPending} onClick={() => parent && make.mutate(parent)}>
        Make variants
      </button>
    </>
  );
}
```

A failure, such as a 409 cycle, names its card in the error toast; the others are still linked.
